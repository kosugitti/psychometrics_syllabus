Example2 <- data.frame(
    id = 1:12,
    num = rep(1:3, 4),
    temp = rep(1:2, each = 6),
    maker = rep(rep(1:2, each = 3), 2),
    value = c(13, 11, 12, 7, 6, 8, 9, 9, 9, 13, 11, 9)
)
Example2$temp <- factor(Example2$temp,
                        labels = c("Hot", "Cold")
)
Example2$maker <- factor(Example2$maker,
                         label = c("A", "B")
)

result.brm2 <- brm(value ~ temp * maker, data = Example2)
result.brm2
\end{lstlisting}
