\begin{lstlisting}[language=c++,label={code:27_08},caption={Creating data and Bayesian estimation with brm}]
Example3 <- data.frame(
  ID = rep(1:4,3),
  time = rep(1:3,each=4),
  value = c(10, 9, 4, 7,5, 4, 2, 3,9, 5, 3, 5)
)
Example3$time <- factor(Example3$time,labels = c("time1","time2","time3"))

result.brm3 <- brm(value ~ time + (1 | ID), data = Example3)
result.brm3
conditional_effects(result.brm3)
\end{lstlisting}
