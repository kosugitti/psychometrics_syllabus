     & = \frac{1}{n}\sum_{i=1}^n (  (t_i - \bar{t}) + (e_i - \bar{e}) )^2                                                                                & \mbox{Combining same symbols}      \\
     & = \frac{1}{n}\sum_{i=1}^n (  (t_i - \bar{t})^2 + 2(t_i - \bar{t})(e_i - \bar{e}) + (e_i - \bar{e})^2  )                                           & \mbox{Expanding}                \\
     & = \frac{1}{n}\sum_{i=1}^n (t_i - \bar{t})^2 +\frac{1}{n}\sum_{i=1}^n 2(t_i - \bar{t})(e_i - \bar{e})  + \frac{1}{n}\sum_{i=1}^n (e_i - \bar{e})^2 & \mbox{Distributing $\sum$}            \\
\end{align*}
