\begin{lstlisting}[language=c++,label={code:27_03},caption={Trying regression analysis}]
# Input data
highschool <- c(2.13,2.42,2.26,3.87,3.90,2.43,3.44,
2.15,2.18,3.00,3.42,2.55,3.19,3.05,2.52)
university <- c(460,500,473,620,690,512,582,
550,485,650,593,528,585,569,518)
# Create data frame
grades <- data.frame(highschool,university)

# Visualize the data
ggplot(grades,aes(x = highschool, y = university)) +
    geom_point() +
    geom_smooth(method = "lm", se = F)

# Linear model using lm function. Write dependent variable ~ explanatory variable
fit <- lm(university ~ highschool, data = grades)
# Display the results in an easy-to-understand way
summary(fit)
\end{lstlisting}
