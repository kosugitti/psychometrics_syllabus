\begin{lstlisting}[language=c++,label={code:08_01},caption={Example of using functions related to probability}]
    dnorm(x = 0, mean = 0, sd = 1)  // Probability density function for normal distribution with mean 0 and standard deviation 1 at x = 0.
    pnorm(q = 1.0, mean = 0, sd = 1)  // Cumulative distribution function for normal distribution with mean 0 and standard deviation 1 up to q = 1.0.
    qnorm(p = 0.618, mean = 0, sd = 1)  // Inverse cumulative distribution function for normal distribution with mean 0 and standard deviation 1 for which the probability up to the returned value is p = 0.618.
    rnorm(n = 10, mean = 0, sd = 1)  // n random numbers generated from normal distribution with mean 0 and standard deviation 1.
\end{lstlisting}
