\begin{lstlisting}[language=c++,label={code:20_01},caption={繰り返しサンプリングする}]
# Clear workspace
rm(list=ls())
### 母集団を作る
X <- rnorm(100,5,1)
## 母平均
mean(X)
### 標本1を作る
s1 <- sample(X,5)
## 標本平均
mean(s1)
## 以下繰り返し
s2 <- sample(X,5)
mean(s2)
s3 <- sample(X,5)
mean(s3)
s4 <- sample(X,5)
mean(s4)
s5 <- sample(X,6)
mean(s5)
\end{lstlisting}
