\begin{lstlisting}[language=c++,label={code:22_03},caption={対応ある二群の仮想データ}]
mu <- c(50,50)
sd1 <- 10
sd2 <- 5
rho <- 0.7
SIG <- matrix(ncol=2,nrow=2)
SIG[1,1] <- sd1 * sd1
SIG[2,2] <- sd2 * sd2
SIG[1,2] <- sd1 * sd2 * rho
SIG[2,1] <- sd1 * sd2 * rho
N <- 100
library(MASS)
X <- mvrnorm(N,mu,SIG)
# Stanに与えるデータセット
dataSet <- list(N=N, X=X)
\end{lstlisting}
