\begin{lstlisting}[language=C++,label={code:x15_01},caption={計量MDSの実践と描画}]
library(ggrepel)
# MDSを実行
result.MDS1 <- cmdscale(eurodist,k=3)
# y軸反転させつつ描画
g <- result.MDS1 %>%
  as.data.frame() %>%
  dplyr::mutate(label = rownames(.)) %>%
  ggplot(aes(x = V1, y = V2, label = label)) +
  geom_point() +
  geom_text_repel() +
  xlim(-2500, 2500) +
  ylim(2500, -2500) +
  xlab("dim 1") +
  ylab("dim2")
\end{lstlisting}
