\begin{lstlisting}[language=c++,label={code:17_06},caption={確率分布の要約}]
N <- 10000
X <- rnorm(N, mean = 0, sd = 1) %>%
  as.data.frame() %>%
  dplyr::rename(val = 1)
X %>%
  summarise(
    Exp = mean(val),
    SD = sd(val),
    Median = median(val),
    U50 = quantile(val, prob = 0.50),
    U90 = quantile(val, prob = 0.90),
    L90 = quantile(val, prob = 0.10)
  )
\end{lstlisting}
