\begin{equation*}
    \bm{a} \bm{b} =
    \begin{pmatrix}
        a_1    \\
        a_2    \\
        \vdots \\
        a_n    \\
    \end{pmatrix}
    \begin{pmatrix}
        b_1 & b_2 & \cdots & b_n \\
    \end{pmatrix}
    =
    \begin{pmatrix}
        a_1 b_1 & a_1 b_2 & \cdots & a_1 b_n \\
        a_2 b_1 & a_2 b_2 & \cdots & a_2 b_n \\
        \vdots  & \vdots  &        & \vdots  \\
        a_n b_1 & a_2 b_2 & \cdots & a_n b_n \\
    \end{pmatrix}
\end{equation*}
