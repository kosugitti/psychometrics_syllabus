\begin{equation*}
    \begin{pmatrix}
        1 & 2 \\
        3 & 4
    \end{pmatrix}
    +
    \begin{pmatrix}
        5 & 6 \\
        7 & 8
    \end{pmatrix}
    =
    \begin{pmatrix}
        1+5 & 2+6 \\
        3+7 & 4+8
    \end{pmatrix}
    =
    \begin{pmatrix}
        6  & 8  \\
        10 & 12
    \end{pmatrix}
\end{equation*}
