library(tidyverse)
dataset <- read_csv("baseballDecade.csv") %>%
  dplyr::filter(Year=="2020年度") %>% 
  dplyr::filter(position == "投手") %>%
  dplyr::select(Name, team, height, weight, salary, Win, Save) %>%
  na.omit() %>%
  arrange(-Win) %>% 
  select(height, weight, salary) %>%
  as.matrix()
